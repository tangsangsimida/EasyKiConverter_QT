65void ExportTargetModel::loadPlugins(const QString& configPath) {
66 QFile file(configPath);
67 if (!file.open(QIODevice::ReadOnly)) {
68 qWarning() << "ExportTargetModel: Failed to open plugin config:" << configPath;
69 return;
70 }
71
72 QByteArray data = file.readAll();
73 file.close();
74
75 QJsonDocument doc = QJsonDocument::fromJson(data);
76 if (doc.isNull() || !doc.isObject()) {
77 qWarning() << "ExportTargetModel: Invalid JSON in plugin config:" << configPath;
78 return;
79 }
80
81 QJsonObject root = doc.object();
82 QJsonArray plugins = root["plugins"].toArray();
83
84 m_targets.clear();
85 for (const QJsonValue& val : plugins) {
86 QJsonObject obj = val.toObject();
87 TargetInfo info;
88 info.id = obj["id"].toString();
89 info.displayName = obj["displayName"].toString();
90 info.icon = obj["icon"].toString();
91 info.optionsComponent = obj["optionsComponent"].toString();
92 if (!info.id.isEmpty() && !info.displayName.isEmpty()) {
93 m_targets.append(info);
94 }
95 }
96
97 // 重建缓存
98 m_availableTargetsCache.clear();
99 for (const TargetInfo& info : m_targets) {
100 QVariantMap map;
101 map["id"] = info.id;
102 map["displayName"] = info.displayName;
103 map["icon"] = info.icon;
104 map["optionsComponent"] = info.optionsComponent;
105 m_availableTargetsCache.append(map);
106 }
107
108 // 确保索引有效
109 bool indexChanged = false;
110 if (m_currentIndex >= m_targets.size()) {
111 m_currentIndex = 0;
112 indexChanged = true;
113 }
114
115 emit availableTargetsChanged();
116 if (indexChanged) {
117 emit currentTargetChanged();
118 }
119}
