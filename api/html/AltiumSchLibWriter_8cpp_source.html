73bool AltiumSchLibWriter::write(const QList<AltiumSchComponent>& components,
74 const QString& filePath,
75 const QString& libraryName) {
76 if (components.isEmpty()) {
77 qWarning() << "AltiumSchLibWriter: Refusing to write an empty library";
78 return false;
79 }
80 m_fonts.clear();
81 m_uniqueIdCounter = 0;
82 m_libraryName = libraryName;
83
84 // 确保有默认字体
85 getOrAddFont("Times New Roman", 10);
86
87 OLECompoundWriter ole;
88 if (!ole.create()) {
89 qWarning() << "AltiumSchLibWriter: Failed to create OLE compound document for" << filePath;
90 return false;
91 }
92
93 QStringList names;
94 for (const AltiumSchComponent& component : components) {
95 names.append(component.name);
96 }
97 const QStringList sectionKeys = AltiumWriterUtils::makeUniqueSectionKeys(names);
98
99 // 写入 FileHeader
100 writeFileHeader(ole, components);
101
102 // 写入 SectionKeys（如果需要）
103 writeSectionKeys(ole, components, sectionKeys);
104
105 // 写入每个元件的存储
106 for (int i = 0; i < components.size(); ++i) {
107 writeComponentStorage(ole, components[i], sectionKeys[i]);
108 }
109
110 // 即使没有图标，也写入合法的 icon-storage 参数记录。部分 AD 版本
111 // 会主动读取该流，零长度占位符并不是 SchLib 的规范表示。
112 QByteArray storageData;
113 AltiumBinaryWriter storageWriter(storageData);
114 QMap<QString, QString> storageParams;
115 storageParams["HEADER"] = "Icon storage";
116 storageWriter.writeCStringParameterBlock(storageParams);
117 ole.writeStream("Storage", storageData);
118
119 return ole.saveToFile(filePath);
120}

125void AltiumSchLibWriter::writeFileHeader(OLECompoundWriter& ole, const QList<AltiumSchComponent>& components) {
126 QMap<QString, QString> params;
127 params["HEADER"] = "Protel for Windows - Schematic Library Editor Binary File Version 5.0";
128 int totalWeight = 0;
129 for (const AltiumSchComponent& component : components) {
130 totalWeight += componentRecordCount(component);
131 }
132 params["WEIGHT"] = QString::number(totalWeight);
133 params["MINORVERSION"] = "2";
134
135 // 生成 8 字符随机 UniqueID
136 QString uid;
137 for (int i = 0; i < 8; ++i) {
138 uid += QChar('A' + QRandomGenerator::global()->bounded(26));
139 }
140 params["UniqueID"] = uid;
141
142 // 字体表
143 params["FontIdCount"] = QString::number(m_fonts.size());
144 for (int i = 0; i < m_fonts.size(); ++i) {
145 int idx = i + 1;
146 params[QString("FontName%1").arg(idx)] = m_fonts[i].name;
147 params[QString("Size%1").arg(idx)] = QString::number(m_fonts[i].size);
148 if (m_fonts[i].bold)
149 params[QString("Bold%1").arg(idx)] = "T";
150 if (m_fonts[i].italic)
151 params[QString("Italic%1").arg(idx)] = "T";
152 if (m_fonts[i].underline)
153 params[QString("Underline%1").arg(idx)] = "T";
154 }
155
156 params["UseMBCS"] = "T";
157 params["IsBOC"] = "T";
158 params["SheetStyle"] = "9";
159 params["BorderOn"] = "T";
160 params["Display_Unit"] = "0";
161 params["SYSTEMFONT"] = "1";
162 params["SHEETNUMBERSPACESIZE"] = "12";
163 params["AREACOLOR"] = "16317695";
164 params["SNAPGRIDON"] = "T";
165 params["SNAPGRIDSIZE"] = "10";
166 params["VISIBLEGRIDON"] = "T";
167 params["VISIBLEGRIDSIZE"] = "10";
168 params["COMPCOUNT"] = QString::number(components.size());
169 for (int i = 0; i < components.size(); ++i) {
170 const AltiumSchComponent& component = components[i];
171 params[QString("LIBREF%1").arg(i)] = component.name;
172 params[QString("COMPDESCR%1").arg(i)] = component.description;
173 // 磁盘格式把公共 Part 0 计入总数，IR 的 partCount 只表示用户可见部件。
174 params[QString("PARTCOUNT%1").arg(i)] = QString::number(qMax(1, component.partCount) + 1);
175 }
176
177 // 序列化
178 QByteArray headerData;
179 AltiumBinaryWriter writer(headerData);
180
181 // Header 中的库索引同样允许非 ASCII 名称/描述，统一用 ANSI fallback + %UTF8% 扩展。
182 writer.writeCStringParameterBlockUtf8(params);
183
184 // 写入元件数量
185 writer.writeInt32(static_cast<int32_t>(components.size()));
186
187 // 写入每个元件名称
188 for (const AltiumSchComponent& comp : components) {
189 writer.writeStringBlock(comp.name);
190 }
191
192 ole.writeStream("FileHeader", headerData);
193}

225void AltiumSchLibWriter::writeComponentStorage(OLECompoundWriter& ole,
226 const AltiumSchComponent& component,
227 const QString& sectionKey) {
228 // 创建存储区
229 ole.addStorage(sectionKey);
230
231 // 构建 Data 流
232 QByteArray data;
233 AltiumBinaryWriter writer(data);
234
235 // 写入元件记录
236 writeComponentRecord(writer, component);
237
238 // 写入引脚
239 for (const AltiumSchPin& pin : component.pins) {
240 writePinRecord(writer, pin, 1);
241 }
242
243 // 写入矩形
244 for (const AltiumSchRectangle& rect : component.rectangles) {
245 writeRectangleRecord(writer, rect);
246 }
247
248 // 写入线段
249 for (const AltiumSchLine& line : component.lines) {
250 writeLineRecord(writer, line);
251 }
252
253 // 写入弧线
254 for (const AltiumSchArc& arc : component.arcs) {
255 writeArcRecord(writer, arc);
256 }
257
258 // 写入多边形
259 for (const AltiumSchPolygon& polygon : component.polygons) {
260 writePolygonRecord(writer, polygon);
261 }
262
263 // 写入椭圆
264 for (const AltiumSchEllipse& ellipse : component.ellipses) {
265 writeEllipseRecord(writer, ellipse);
266 }
267
268 // 写入折线
269 for (const AltiumSchPolyline& polyline : component.polylines) {
270 writePolylineRecord(writer, polyline);
271 }
272
273 for (const AltiumSchPath& path : component.paths) {
274 writePathRecord(writer, path);
275 }
276
277 // 写入文本
278 for (const AltiumSchText& text : component.texts) {
279 writeTextRecord(writer, text);
280 }
281
282 writeComponentParameterRecords(writer, component);
283
284 // 写入实现记录
285 writeImplementationRecords(writer, component);
286
287 // 写入流
288 ole.writeStream(sectionKey, "Data", data);
289}

321void AltiumSchLibWriter::writePinRecord(AltiumBinaryWriter& writer, const AltiumSchPin& pin, int partId) {
322 writer.beginBlock(AltiumConstants::SCH_BLOCK_FLAG_BINARY_PIN);
323
324 writer.writeInt32(2); // Record type = 2
325 writer.writeUInt8(0); // Unknown
326 Q_UNUSED(partId);
327 writer.writeInt16(static_cast<int16_t>(qBound(1, pin.ownerPartId, 32767))); // OwnerPartId
328 writer.writeUInt8(0); // OwnerPartDisplayMode
329
330 // Symbol edges / IEEE 装饰。四个字节必须位于描述字符串之前。
331 writer.writeUInt8(pin.symbolInnerEdge);
332 writer.writeUInt8(pin.symbolOuterEdge);
333 writer.writeUInt8(pin.symbolInside);
334 writer.writeUInt8(pin.symbolOutside);
335
336 // Description (空 Pascal 短字符串)
337 writer.writePascalShortString("");
338
339 // FormalType=1 表示普通的有效引脚。值为 0 时，Altium 仍可能绘制
340 // 引脚名称/编号，但不会将该记录作为可连接的 Pin 对象处理。
341 writer.writeUInt8(1); // FormalType: normal pin
342 writer.writeUInt8(static_cast<uint8_t>(pin.electricalType)); // ElectricalType
343
344 // PinConglomerate 字节
345 uint8_t conglomerate = static_cast<uint8_t>(pin.orientation); // Bit 0-1: orientation
346 if (pin.isHidden)
347 conglomerate |= 0x04; // Bit 2: hidden
348 if (pin.showName)
349 conglomerate |= 0x08; // Bit 3: show name
350 if (pin.showDesignator)
351 conglomerate |= 0x10; // Bit 4: show designator
352 writer.writeUInt8(conglomerate);
353
354 // PinLength (DXP 整数单位)
355 int16_t lengthDxp = AltiumCoord::toDxpInt(pin.length);
356 writer.writeInt16(lengthDxp);
357
358 // Location (DXP 整数单位)
359 writer.writeInt16(AltiumCoord::toDxpInt(pin.locationX));
360 writer.writeInt16(AltiumCoord::toDxpInt(pin.locationY));
361
362 // Color
363 writer.writeUInt32(pin.color);
364
365 // Name (Pascal 短字符串)
366 writer.writePascalShortString(pin.name);
367
368 // Designator (Pascal 短字符串)
369 writer.writePascalShortString(pin.designator);
370
371 // SwapIdGroup (空 Pascal 短字符串)
372 writer.writePascalShortString("");
373
374 // PartAndSequence (空 Pascal 短字符串)
375 writer.writePascalShortString("");
376
377 // DefaultValue (空 Pascal 短字符串)
378 writer.writePascalShortString("");
379
380 writer.endBlock();
381}

609void AltiumSchLibWriter::writeImplementationRecords(AltiumBinaryWriter& writer, const AltiumSchComponent& component) {
610 // RECORD=44: ImplementationList（容器，始终写入）
611 {
612 QMap<QString, QString> params;
613 params["RECORD"] = "44";
614 params["DataFileFormatID"] = "";
615 params["Description"] = "";
616 params["FileName"] = "";
617 writer.writeCStringParameterBlock(params);
618 }
619
620 // 每个实现
621 for (const AltiumSchComponent::Implementation& impl : component.implementations) {
622 // RECORD=45: Implementation
623 {
624 QMap<QString, QString> params;
625 params["RECORD"] = "45";
626 params["DESCRIPTION"] = impl.modelName;
627 params["MODELNAME"] = impl.modelName;
628 params["MODELTYPE"] = impl.modelType;
629 params["DATAFILECOUNT"] = "1";
630 params["MODELDATAFILEKIND1"] = "PCBLib";
631 params["MODELDATAFILEENTITY1"] = m_libraryName.isEmpty() ? "*" : m_libraryName + ".PcbLib";
632 params["ISCURRENT"] = "T";
633 addUniqueID(params);
634 writer.writeCStringParameterBlock(params);
635 }
636
637 // RECORD=46: MapDefinerList（容器）
638 {
639 QMap<QString, QString> params;
640 params["RECORD"] = "46";
641 writer.writeCStringParameterBlock(params);
642 }
643
644 // RECORD=47: 将每个引脚映射到当前实现。
645 // Altium 会依据这些索引建立符号引脚与 PCBLib 实现之间的关系；
646 // 缺少该记录时，库虽然可以打开，但引脚归属和封装关联并不完整。
647 for (int pinIndex = 1; pinIndex <= component.pins.size(); ++pinIndex) {
648 QMap<QString, QString> pinMappingParams;
649 pinMappingParams["RECORD"] = "47";
650 pinMappingParams["DESINTF"] = QString::number(pinIndex);
651 pinMappingParams["DESIMPCOUNT"] = "1";
652 pinMappingParams["DESIMP0"] = QString::number(pinIndex);
653 pinMappingParams["ISTRIVIAL"] = "T";
654 addUniqueID(pinMappingParams);
655 writer.writeCStringParameterBlock(pinMappingParams);
656 }
657
658 // RECORD=48: ImplementationParameters（空容器）
659 {
660 QMap<QString, QString> params;
661 params["RECORD"] = "48";
662 writer.writeCStringParameterBlock(params);
663 }
664 }

672int AltiumSchLibWriter::componentRecordCount(const AltiumSchComponent& component) const {
673 const int graphics = component.pins.size() + component.rectangles.size() + component.lines.size() +
674 component.arcs.size() + component.polygons.size() + component.ellipses.size() +
675 component.polylines.size() + component.paths.size() + component.texts.size();
676 // Component + graphics + Designator + Comment + ImplementationList + implementation triplets.
677 return 1 + graphics + 2 + 1 + component.implementations.size() * 3;
678}
