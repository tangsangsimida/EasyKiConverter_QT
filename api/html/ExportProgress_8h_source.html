246 int totalSuccessCount() const {

250 }
251 return count;
252 }
253
258 int totalFailedCount() const {
259 int count = 0;
260 for (const auto& p : exportTypeProgress) {
261 count += p.failedCount;
262 }
263 return count;

265
270 bool isComplete() const {
271 if (currentStage != Stage::Exporting && currentStage != Stage::Preloading) {
272 return currentStage == Stage::Completed || currentStage == Stage::Cancelled ||
273 currentStage == Stage::Failed;
274 }
275 if (currentStage == Stage::Preloading) {
276 return preloadProgress.isComplete();
277 }
278 for (const auto& p : exportTypeProgress) {
279 if (!p.isComplete())
280 return false;
281 }
282 return true;
283 }
284
296 int overallPercentage() const {
297 if (currentStage == Stage::Idle)
298 return 0;
299 if (currentStage == Stage::Preloading)
300 return preloadProgress.percentage();
301 if (currentStage == Stage::Exporting) {
302 int total = 0, completed = 0;
303 for (const auto& p : exportTypeProgress) {
304 total += p.totalCount;
305 completed += p.completedCount;
306 }
307 return total > 0 ? (completed * 100 / total) : 0;
308 }
309 if (currentStage == Stage::Completed)
310 return 100;
311 return 0;
312 }
