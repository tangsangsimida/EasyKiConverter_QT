21uint32_t toAltiumColor(const QColor& color) {
22 // Altium 原理图库的默认符号色是深蓝色；黑色是 IR 的“未指定”默认值，
23 // 若直接写入 0，SchLib 写入器会省略 Color 参数并由 AD 使用不可预测的默认值。
24 constexpr uint32_t kDefaultSchematicColor = 0x68380B;
25 if (!color.isValid() || color.alpha() == 0 || color == QColor(Qt::black))
26 return kDefaultSchematicColor;
27 return (static_cast<uint32_t>(color.blue()) << 16) | (static_cast<uint32_t>(color.green()) << 8) |
28 static_cast<uint32_t>(color.red());
29}

37int quantizeSchematicOffset(int value) {
38 constexpr int kSchematicUnitRaw = 1000;
39 if (value >= 0)
40 return ((value + kSchematicUnitRaw / 2) / kSchematicUnitRaw) * kSchematicUnitRaw;
41 return ((value - kSchematicUnitRaw / 2) / kSchematicUnitRaw) * kSchematicUnitRaw;
42}

49int quantizeToSnapGrid(int value) {
50 constexpr int kSnapGrid = 100000; // 10 mil
51 if (value >= 0)
52 return ((value + kSnapGrid / 2) / kSnapGrid) * kSnapGrid;
53 return ((value - kSnapGrid / 2) / kSnapGrid) * kSnapGrid;
54}

63QPointF computePinConnectionPoint(const AltiumSchPin& pin) {
64 switch (pin.orientation) {
65 case AltiumModels::PinOrientation::Right:
66 return QPointF(pin.locationX + pin.length, pin.locationY);
67 case AltiumModels::PinOrientation::Left:
68 return QPointF(pin.locationX - pin.length, pin.locationY);
69 case AltiumModels::PinOrientation::Up:
70 return QPointF(pin.locationX, pin.locationY + pin.length);
71 case AltiumModels::PinOrientation::Down:
72 return QPointF(pin.locationX, pin.locationY - pin.length);
73 }
74 return QPointF(pin.locationX, pin.locationY);
75}

83void quantizePinConnectionPoint(AltiumSchPin& pin) {
84 QPointF conn = computePinConnectionPoint(pin);
85 int quantizedConnX = quantizeToSnapGrid(static_cast<int>(conn.x()));
86 int quantizedConnY = quantizeToSnapGrid(static_cast<int>(conn.y()));
87
88 switch (pin.orientation) {
89 case AltiumModels::PinOrientation::Right:
90 pin.locationX = quantizedConnX - pin.length;
91 pin.locationY = quantizedConnY;
92 break;
93 case AltiumModels::PinOrientation::Left:
94 pin.locationX = quantizedConnX + pin.length;
95 pin.locationY = quantizedConnY;
96 break;
97 case AltiumModels::PinOrientation::Up:
98 pin.locationX = quantizedConnX;
99 pin.locationY = quantizedConnY - pin.length;
100 break;
101 case AltiumModels::PinOrientation::Down:
102 pin.locationX = quantizedConnX;
103 pin.locationY = quantizedConnY + pin.length;
104 break;
105 }
106}

111void quantizePinConnectionGroups(QList<AltiumSchPin>& pins) {
112 constexpr int kSnapGrid = 100000;
113 for (int side = 0; side < 4; ++side) {
114 QList<int> indices;
115 for (int i = 0; i < pins.size(); ++i) {
116 if (static_cast<int>(pins[i].orientation) == side)
117 indices.append(i);
118 }
119 std::sort(indices.begin(), indices.end(), [&](int lhs, int rhs) {
120 const QPointF a = computePinConnectionPoint(pins[lhs]);
121 const QPointF b = computePinConnectionPoint(pins[rhs]);
122 const bool horizontal = side == static_cast<int>(AltiumModels::PinOrientation::Right) ||
123 side == static_cast<int>(AltiumModels::PinOrientation::Left);
124 return horizontal ? (a.y() < b.y()) : (a.x() < b.x());
125 });
126
127 int previousTangent = INT_MIN;
128 for (int index : indices) {
129 quantizePinConnectionPoint(pins[index]);
130 QPointF conn = computePinConnectionPoint(pins[index]);
131 const bool horizontal = side == static_cast<int>(AltiumModels::PinOrientation::Right) ||
132 side == static_cast<int>(AltiumModels::PinOrientation::Left);
133 int tangent = horizontal ? static_cast<int>(conn.y()) : static_cast<int>(conn.x());
134 if (previousTangent != INT_MIN && tangent <= previousTangent) {
135 tangent = previousTangent + kSnapGrid;
136 switch (pins[index].orientation) {
137 case AltiumModels::PinOrientation::Right:
138 case AltiumModels::PinOrientation::Left:
139 pins[index].locationY = tangent;
140 break;
141 case AltiumModels::PinOrientation::Up:
142 case AltiumModels::PinOrientation::Down:
143 pins[index].locationX = tangent;
144 break;
145 }
146 }
147 previousTangent = tangent;
148 }
149 }
150}

190AltiumSchComponent ExporterAltiumSymbol::convertSymbol(const IR::SymbolComponentIR& data) {
191 AltiumSchComponent component;
192 component.name = data.name;
193 component.description = data.description;
194 component.designatorPrefix = data.designatorPrefix;
195 component.partCount = data.partCount;
196
197 // 转换引脚
198 for (const IR::SymbolPinIR& pin : data.pins) {
199 component.pins.append(convertPin(pin));
200 if (pin.hasNamePosition && !pin.name.isEmpty()) {
201 AltiumSchText text;
202 text.locationX = AltiumCoord::mmToRaw(pin.namePosition.x());
203 text.locationY = AltiumCoord::mmToRaw(pin.namePosition.y());
204 text.text = pin.name;
205 text.fontSizeMm = pin.nameFontSizeMm;
206 text.anchor = pin.nameAnchor;
207 text.isDisplayed = true;
208 text.orientation = static_cast<int>(pin.nameRotation / 90.0) % 4;
209 text.ownerPartId = qMax(1, pin.partIndex + 1);
210 component.texts.append(text);
211 }
212 if (pin.hasNumberPosition && !pin.designator.isEmpty()) {
213 AltiumSchText text;
214 text.locationX = AltiumCoord::mmToRaw(pin.numberPosition.x());
215 text.locationY = AltiumCoord::mmToRaw(pin.numberPosition.y());
216 text.text = pin.designator;
217 text.fontSizeMm = pin.numberFontSizeMm;
218 text.anchor = pin.numberAnchor;
219 text.isDisplayed = true;
220 text.orientation = static_cast<int>(pin.numberRotation / 90.0) % 4;
221 text.ownerPartId = qMax(1, pin.partIndex + 1);
222 component.texts.append(text);
223 }
224 }
225
226 // 转换图形元素
227 for (const IR::SymbolRectangleIR& r : data.rectangles)
228 component.rectangles.append(convertRectangle(r));
229 for (const IR::SymbolCircleIR& c : data.circles)
230 component.ellipses.append(convertCircle(c));
231 for (const IR::SymbolArcIR& a : data.arcs)
232 component.arcs.append(convertArc(a));
233 for (const IR::SymbolPolygonIR& p : data.polygons)
234 component.polygons.append(convertPolygon(p));
235 for (const IR::SymbolPolylineIR& p : data.polylines)
236 component.polylines.append(convertPolyline(p));
237 for (const IR::SymbolPathIR& p : data.paths)
238 component.paths.append(convertPath(p));
239 for (const IR::SymbolTextIR& t : data.texts)
240 component.texts.append(convertText(t));
241 for (const IR::SymbolEllipseIR& e : data.ellipses)
242 component.ellipses.append(convertEllipse(e));
243
244 // 添加封装链接
245 if (!data.footprintName.isEmpty()) {
246 AltiumSchComponent::Implementation impl;
247 impl.modelName = data.footprintName;
248 impl.modelType = "PCBLIB";
249 component.implementations.append(impl);
250 }
251
252 // 坐标归一化：将符号中心移到原点
253 centerComponent(component);
254
255 return component;
256}

261AltiumSchPin ExporterAltiumSymbol::convertPin(const IR::SymbolPinIR& pin) {
262 AltiumSchPin altiumPin;
263 altiumPin.name = pin.name;
264 altiumPin.designator = pin.designator;
265 altiumPin.locationX = AltiumCoord::mmToRaw(pin.position.x());
266 altiumPin.locationY = AltiumCoord::mmToRaw(pin.position.y());
267 altiumPin.length = pin.length > 0.0 ? AltiumCoord::mmToRaw(pin.length) : 100000;
268 altiumPin.electricalType =
269 static_cast<AltiumModels::PinElectricalType>(AltiumLayerMap::toAltiumElectricalType(pin.electricalType));
270 altiumPin.orientation =
271 static_cast<AltiumModels::PinOrientation>(AltiumLayerMap::toAltiumPinOrientation(pin.direction));
272 // EasyEDA 的 pin name 显示标志在部分库中未设置，但名称字符串本身
273 // 仍是符号的一部分。Altium 的 PinConglomerate 必须显式打开 show-name，
274 // 否则 AD 只绘制 Pin Number，名称会表现为脱离引脚的独立文本。
275 // 优先从 display 控制层获取，兼容旧 showName 字段。
276 altiumPin.showName =
277 !pin.hasNamePosition && (pin.display.showName || pin.showName || !pin.name.trimmed().isEmpty());
278 altiumPin.showDesignator = !pin.hasNumberPosition && (pin.display.showDesignator || pin.showDesignator);
279 altiumPin.isHidden = !altiumPin.showName && !altiumPin.showDesignator;
280 altiumPin.color = toAltiumColor(QColor(Qt::black));
281 // EasyEDA 的反相圆点位于引脚外侧，时钟标记贴近主体内侧。
282 // 优先从 style 语义层获取，兼容旧 hasDot/hasClock 字段。
283 if (pin.style.inverted || pin.hasDot)
284 altiumPin.symbolOuterEdge = 1; // Dot
285 if (pin.style.clock || pin.hasClock)
286 altiumPin.symbolInnerEdge = 3; // Clock
287
288 // 处理更丰富的 PinDecoration 枚举
289 switch (pin.style.decoration) {
290 case IR::PinDecoration::OpenCollector:
291 altiumPin.symbolOuterEdge = 8;
292 break;
293 case IR::PinDecoration::OpenEmitter:
294 altiumPin.symbolOuterEdge = 15;
295 break;
296 case IR::PinDecoration::HiZ:
297 altiumPin.symbolOuterEdge = 12;
298 break;
299 case IR::PinDecoration::Pulse:
300 altiumPin.symbolOuterEdge = 14;
301 break;
302 case IR::PinDecoration::Postponed:
303 altiumPin.symbolOuterEdge = 11;
304 break;
305 default:
306 break;
307 }
308
309 // 这些电气类型同时具有明确的 Altium IEEE 装饰
310 switch (pin.electricalType) {
311 case IR::PinElectricalType::OpenCollector:
312 altiumPin.symbolOuterEdge = 8;
313 break;
314 case IR::PinElectricalType::OpenEmitter:
315 altiumPin.symbolOuterEdge = 15;
316 break;
317 default:
318 break;
319 }
320 altiumPin.ownerPartId = qMax(1, pin.partIndex + 1);
321
322 // 电源引脚检测：EasyEDA 通常不区分电源引脚（type=3/Bidirectional），
323 // 通过引脚名称匹配常见电源网络名称，强制设为 Power 类型
324 if (altiumPin.electricalType != AltiumModels::PinElectricalType::Power) {
325 static const QSet<QString> powerPinNames = {
326 "GND", "AGND", "DGND", "PGND", "SGND", "CGND", "GNDP", "GNDN", "VCC",
327 "VDD", "AVCC", "AVDD", "DVDD", "IOVDD", "PVDD", "SVDD", "VDDA", "VDDIO",
328 "VDDS", "VDDP", "VBUS", "VSYS", "VIN", "5V", "3V3", "1V8", "USB_VDD",
329 "ADC_AVDD", "VREG_IN", "VREG_OUT", "VREG_VOUT", "VEE", "VSS", "VSSA",
330 };
331 QString upperName = pin.name.toUpper().trimmed();
332 if (powerPinNames.contains(upperName)) {
333 altiumPin.electricalType = AltiumModels::PinElectricalType::Power;
334 }
335 }
336
337 return altiumPin;
338}

377AltiumSchArc ExporterAltiumSymbol::convertArc(const IR::SymbolArcIR& arc) {
378 AltiumSchArc altiumArc;
379 // 三点确定圆弧。退化为共线时，以首尾中点作为安全回退。
380 const double ax = arc.startPoint.x();
381 const double ay = arc.startPoint.y();
382 const double bx = arc.midPoint.x();
383 const double by = arc.midPoint.y();
384 const double cx = arc.endPoint.x();
385 const double cy = arc.endPoint.y();
386 const double determinant = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by));
387 QPointF center = (arc.startPoint + arc.endPoint) / 2.0;
388 if (std::abs(determinant) > 1e-12) {
389 const double a2 = ax * ax + ay * ay;
390 const double b2 = bx * bx + by * by;
391 const double c2 = cx * cx + cy * cy;
392 center.setX((a2 * (by - cy) + b2 * (cy - ay) + c2 * (ay - by)) / determinant);
393 center.setY((a2 * (cx - bx) + b2 * (ax - cx) + c2 * (bx - ax)) / determinant);
394 }
395 double dx = arc.startPoint.x() - center.x();
396 double dy = arc.startPoint.y() - center.y();
397 double radius = std::sqrt(dx * dx + dy * dy);
398 altiumArc.centerX = AltiumCoord::mmToRaw(center.x());
399 altiumArc.centerY = AltiumCoord::mmToRaw(center.y());
400 altiumArc.radius = AltiumCoord::mmToRaw(radius);
401 altiumArc.startAngle = std::atan2(arc.startPoint.y() - center.y(), arc.startPoint.x() - center.x()) * 180.0 / M_PI;
402 altiumArc.endAngle = std::atan2(arc.endPoint.y() - center.y(), arc.endPoint.x() - center.x()) * 180.0 / M_PI;
403 altiumArc.lineWidth = AltiumCoord::lineWidthMmToIndex(arc.strokeWidth);
404 altiumArc.color = toAltiumColor(arc.strokeColor);
405 altiumArc.ownerPartId = qMax(1, arc.partIndex + 1);
406 return altiumArc;
407}

497void ExporterAltiumSymbol::centerComponent(AltiumSchComponent& component) {
498 if (component.pins.isEmpty())
499 return;
500
501 int minX = INT_MAX, minY = INT_MAX, maxX = INT_MIN, maxY = INT_MIN;
502 auto include = [&](int x, int y) {
503 minX = qMin(minX, x);
504 minY = qMin(minY, y);
505 maxX = qMax(maxX, x);
506 maxY = qMax(maxY, y);
507 };
508 for (const auto& pin : component.pins)
509 include(pin.locationX, pin.locationY);
510 for (const auto& rect : component.rectangles) {
511 include(rect.locationX, rect.locationY);
512 include(rect.cornerX, rect.cornerY);
513 }
514 for (const auto& arc : component.arcs) {
515 include(arc.centerX - arc.radius, arc.centerY - arc.radius);
516 include(arc.centerX + arc.radius, arc.centerY + arc.radius);
517 }
518 for (const auto& ellipse : component.ellipses) {
519 include(ellipse.centerX - ellipse.radiusX, ellipse.centerY - ellipse.radiusY);
520 include(ellipse.centerX + ellipse.radiusX, ellipse.centerY + ellipse.radiusY);
521 }
522 if (minX == INT_MAX)
523 return;
524
525 // 量化平移量而不是逐个量化引脚，保持所有相对间距不变。
526 int offsetX = quantizeSchematicOffset((minX + maxX) / 2);
527 int offsetY = quantizeSchematicOffset((minY + maxY) / 2);
528 // Altium 符号的可编辑原点相对 EasyEDA 几何中心有一个 10 mil 的
529 // Y 基准偏置。将偏置加入统一平移量，使主体上下边界和四侧引脚
530 // 在同一坐标基准上（例如 C2040 的 -180..200 mil）。
531 constexpr int kAltiumSymbolOriginYBias = 1000000;
532 offsetY -= kAltiumSymbolOriginYBias;
533 int offsetPolyX = offsetX / 1000;
534 int offsetPolyY = offsetY / 1000;
535
536 // 平移引脚
537 for (auto& pin : component.pins) {
538 pin.locationX -= offsetX;
539 pin.locationY -= offsetY;
540 }
541 // 平移矩形
542 for (auto& rect : component.rectangles) {
543 rect.locationX -= offsetX;
544 rect.locationY -= offsetY;
545 rect.cornerX -= offsetX;
546 rect.cornerY -= offsetY;
547 }
548 // 平移弧线
549 for (auto& arc : component.arcs) {
550 arc.centerX -= offsetX;
551 arc.centerY -= offsetY;
552 }
553 // 平移椭圆
554 for (auto& ellipse : component.ellipses) {
555 ellipse.centerX -= offsetX;
556 ellipse.centerY -= offsetY;
557 }
558 // 平移多边形（mmToSchematicUnits 坐标系）
559 for (auto& poly : component.polygons) {
560 for (QPointF& v : poly.vertices) {
561 v = QPointF(v.x() - offsetPolyX, v.y() - offsetPolyY);
562 }
563 }
564 for (auto& polyline : component.polylines) {
565 for (QPointF& v : polyline.vertices) {
566 v = QPointF(v.x() - offsetPolyX, v.y() - offsetPolyY);
567 }
568 }
569 for (auto& path : component.paths) {
570 for (QPointF& v : path.vertices) {
571 v = QPointF(v.x() - offsetPolyX, v.y() - offsetPolyY);
572 }
573 }
574 // 平移文本
575 for (auto& text : component.texts) {
576 text.locationX -= offsetX;
577 text.locationY -= offsetY;
578 }
579
580 // 统一引脚与主体的法向锚点。Altium 的 Pin Name/Number 都以二进制
581 // 引脚位置为基准；如果引脚位置落在主体边界外，名称就会看起来与图形
582 // 脱离。仅修正朝向法向坐标，沿边方向坐标保持不变，因此不会改变引脚
583 // 间距。没有矩形主体的符号不强制投影，避免破坏原始几何。
584 if (!component.rectangles.isEmpty()) {
585 int bodyMinX = INT_MAX, bodyMinY = INT_MAX;
586 int bodyMaxX = INT_MIN, bodyMaxY = INT_MIN;
587 for (const auto& rect : component.rectangles) {
588 bodyMinX = qMin(bodyMinX, qMin(rect.locationX, rect.cornerX));
589 bodyMinY = qMin(bodyMinY, qMin(rect.locationY, rect.cornerY));
590 bodyMaxX = qMax(bodyMaxX, qMax(rect.locationX, rect.cornerX));
591 bodyMaxY = qMax(bodyMaxY, qMax(rect.locationY, rect.cornerY));
592 }
593 for (auto& pin : component.pins) {
594 switch (pin.orientation) {
595 case AltiumModels::PinOrientation::Right:
596 pin.locationX = bodyMaxX;
597 break;
598 case AltiumModels::PinOrientation::Left:
599 pin.locationX = bodyMinX;
600 break;
601 case AltiumModels::PinOrientation::Up:
602 pin.locationY = bodyMaxY;
603 break;
604 case AltiumModels::PinOrientation::Down:
605 pin.locationY = bodyMinY;
606 break;
607 }
608 }
609
610 // 将引脚连接端量化到 10 mil 吸附网格，保持引脚长度不变。
611 quantizePinConnectionGroups(component.pins);
612
613 // EasyEDA 的 Value/Comment 字段有时共享同一个锚点，并携带 270°
614 // 的画布旋转值。直接写入 Altium 后，两个字段会被放到符号外部或
615 // 变成不可见的竖排文本。仅当多个可见字段确实重合时，按 Altium
616 // 符号字段习惯在主体中心水平排列；独立文本仍保留其原始几何。
617 QList<int> visibleTextIndices;
618 for (int i = 0; i < component.texts.size(); ++i) {
619 if (!component.texts[i].isHidden)
620 visibleTextIndices.append(i);
621 }
622 if (visibleTextIndices.size() > 1) {
623 const auto& first = component.texts[visibleTextIndices.first()];
624 bool coincident = true;
625 for (int index : visibleTextIndices) {
626 const auto& text = component.texts[index];
627 if (qAbs(text.locationX - first.locationX) > 10000 || qAbs(text.locationY - first.locationY) > 10000) {
628 coincident = false;
629 break;
630 }
631 }
632 if (coincident) {
633 const int centerX = (bodyMinX + bodyMaxX) / 2;
634 const int centerY = (bodyMinY + bodyMaxY) / 2;
635 constexpr int kFieldStart = -300000; // 主体中心 + (-3 mil) = 7 mil
636 constexpr int kFieldSpacing = 2000000; // 20 mil
637 for (int order = 0; order < visibleTextIndices.size(); ++order) {
638 auto& text = component.texts[visibleTextIndices[order]];
639 text.locationX = centerX;
640 text.locationY = centerY + kFieldStart + order * kFieldSpacing;
641 text.orientation = 0;
642 }
643 }
644 }
645 }
646}
