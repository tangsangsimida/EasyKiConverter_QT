47bool AltiumPcbLibWriter::write(const QList<AltiumPcbComponent>& components,
48 const QString& filePath,
49 const QString& libraryName) {
50 if (components.isEmpty()) {
51 qWarning() << "AltiumPcbLibWriter: Refusing to write an empty library";
52 return false;
53 }
54 m_wideStrings.clear();
55
56 qDebug() << "AltiumPcbLibWriter::write: components:" << components.size() << "filePath:" << filePath;
57
58 OLECompoundWriter ole;
59 if (!ole.create()) {
60 qWarning() << "AltiumPcbLibWriter::write: Failed to create OLE compound document";
61 return false;
62 }
63
64 QStringList names;
65 for (const AltiumPcbComponent& component : components) {
66 names.append(component.name);
67 }
68 const QStringList sectionKeys = AltiumWriterUtils::makeUniqueSectionKeys(names);
69
70 // 只输出 PcbLib 核心流。未知或不完整的可选 storage 比缺省更危险，
71 // 因为 Altium 会尝试按其声明的版本反序列化它们。
72 writeFileHeader(ole);
73 writeSectionKeys(ole, components, sectionKeys);
74 writeLibraryStorage(ole, components, filePath);
75
76 // 写入每个封装
77 for (int i = 0; i < components.size(); ++i) {
78 writeFootprintStorage(ole, components[i], sectionKeys[i]);
79 }
80
81 bool result = ole.saveToFile(filePath);
82 qDebug() << "AltiumPcbLibWriter::write: saveToFile result:" << result << "filePath:" << filePath;
83 return result;
84}

90 QByteArray header;

106 const QList<AltiumPcbComponent>& components,

108 QByteArray data;

110
111 // 计算需要映射的条目数
112 QVector<QPair<QString, QString>> mappings;
113 for (int i = 0; i < components.size(); ++i) {
114 const AltiumPcbComponent& comp = components[i];
115 const QString& key = sectionKeys[i];
116 if (key != comp.name) {
117 mappings.append({comp.name, key});
118 }
119 }
120
121 writer.writeUInt32(static_cast<uint32_t>(mappings.size()));
122 for (const auto& mapping : mappings) {
123 writer.writePascalString(mapping.first);
124 writer.writeStringBlock(mapping.second);
125 }
126
127 if (!mappings.isEmpty()) {
128 ole.writeStream("SectionKeys", data);

199 auto safeValue = [](QString value) {
200 value.replace('|', ' ');
201 value.replace('\0', ' ');
202 return value;
203 };
204 auto append = [&safeValue](QStringList& fields, const QString& key, const QString& value) {
205 fields.append(key + "=" + safeValue(value));
206 };
207
208 QString absolutePath = QFileInfo(filePath).absoluteFilePath();
209 absolutePath.replace('/', '\\');
210 const QDateTime now = QDateTime::currentDateTime();
211
212 // Altium serializes library settings as a sequence of Board records. We build
213 // that profile from typed defaults so it remains reviewable and evolvable
214 // instead of embedding an opaque captured blob.
215 QList<QStringList> records;
216 records.append(QStringList{});
217 QStringList& identity = records.last();
218 append(identity, "FILENAME", absolutePath);
219 append(identity, "KIND", "Protel_Advanced_PCB_Library");
220 append(identity, "VERSION", "3.00");
221 append(identity, "DATE", QLocale::c().toString(now.date(), "M/d/yyyy"));
222 append(identity, "TIME", QLocale::c().toString(now.time(), "h:mm:ss AP"));
223 append(identity, "TOPTYPE", "3");
224 append(identity, "TOPCONST", "3.500");
225 append(identity, "TOPHEIGHT", "0.4mil");
226 append(identity, "TOPMATERIAL", "Solder Resist");
227 append(identity, "BOTTOMTYPE", "3");
228 append(identity, "BOTTOMCONST", "3.500");
229 append(identity, "BOTTOMHEIGHT", "0.4mil");
230 append(identity, "BOTTOMMATERIAL", "Solder Resist");
231 append(identity, "LAYERSTACKSTYLE", "0");
232
233 auto layerName = [](int layer) -> QString {
234 if (layer == 1)
235 return "Top Layer";
236 if (layer >= 2 && layer <= 31)
237 return QString("Mid-Layer %1").arg(layer - 1);
238 if (layer == 32)
239 return "Bottom Layer";
240 static const QStringList standardNames = {
241 "Top Overlay", "Bottom Overlay", "Top Paste", "Bottom Paste", "Top Solder", "Bottom Solder"};
242 if (layer >= 33 && layer <= 38)
243 return standardNames[layer - 33];
244 if (layer >= 39 && layer <= 54)
245 return QString("Internal Plane %1").arg(layer - 38);
246 if (layer == 55)
247 return "Drill Guide";
248 if (layer == 56)
249 return "Keep-Out Layer";
250 if (layer >= 57 && layer <= 72)
251 return QString("Mechanical %1").arg(layer - 56);
252 static const QStringList auxiliaryNames = {"Drill Drawing",
253 "Multi-Layer",
254 "Connections",
255 "Background",
256 "DRC Error Markers",
257 "Selections",
258 "Visible Grid 1",
259 "Visible Grid 2",
260 "Pad Holes",
261 "Via Holes"};
262 return auxiliaryNames.value(layer - 73, QString("Layer %1").arg(layer));
263 };
264
265 for (int layer = 1; layer <= 82; ++layer) {
266 if ((layer - 1) % 5 == 0 && layer != 1) {
267 records.append(QStringList{});
268 }
269 QStringList& fields = records.last();
270 const QString prefix = QString("LAYER%1").arg(layer);
271 append(fields, prefix + "NAME", layerName(layer));
272 append(fields, prefix + "PREV", layer == 1 ? "0" : (layer == 32 ? "1" : "0"));
273 append(fields, prefix + "NEXT", layer == 1 ? "32" : "0");
274 append(fields, prefix + "MECHENABLED", layer == 57 ? "TRUE" : "FALSE");
275 append(fields, prefix + "COPTHICK", "1.4mil");
276 append(fields, prefix + "DIELTYPE", "0");
277 append(fields, prefix + "DIELCONST", "4.800");
278 append(fields, prefix + "DIELHEIGHT", "12.6mil");
279 append(fields, prefix + "DIELMATERIAL", "FR-4");
280 }
281
282 records.append(QStringList{});
283 QStringList& grid = records.last();
284 append(grid, "BIGVISIBLEGRIDSIZE", "0.000000");
285 append(grid, "VISIBLEGRIDSIZE", "0.000000");
286 append(grid, "SNAPGRIDSIZE", "50000.000000");
287 append(grid, "SNAPGRIDSIZEX", "50000.000000");
288 append(grid, "SNAPGRIDSIZEY", "50000.000000");
289 append(grid, "ELECTRICALGRIDRANGE", "8mil");
290 append(grid, "ELECTRICALGRIDENABLED", "TRUE");
291 append(grid, "DISPLAYUNIT", "1");
292 append(grid, "CURRENT2D3DVIEWSTATE", "2D");
293
294 QStringList serializedRecords;
295 for (const QStringList& record : records) {
296 if (!record.isEmpty()) {
297 serializedRecords.append(record.join('|'));
298 }
299 }
300 return serializedRecords.join("\r|RECORD=Board|");
301}

306void AltiumPcbLibWriter::writeModelsStorage(OLECompoundWriter& ole, const QList<AltiumPcbComponent>& components) {
307 ole.addStorage("Library", "Models");
308
309 // 收集所有 3D 模型
310 QList<AltiumPcbComponent::Model3D> allModels;
311 for (const AltiumPcbComponent& comp : components) {
312 for (const AltiumPcbComponent::Model3D& model : comp.models) {
313 allModels.append(model);
314 }
315 }
316
317 // Header 流
318 QByteArray hdrData;
319 AltiumBinaryWriter hdr(hdrData);
320 hdr.writeUInt32(static_cast<uint32_t>(allModels.size()));
321 ole.writeStream("Library/Models", "Header", hdrData);
322
323 // Data 流（模型元数据）
324 QByteArray modelData;
325 AltiumBinaryWriter modelWriter(modelData);
326 for (const AltiumPcbComponent::Model3D& model : allModels) {
327 const QString id = model.id.isEmpty() ? model.name : model.id;
328 const QString metadata =
329 QString("EMBED=TRUE|MODELSOURCE=Undefined|ID=%1|ROTX=%2|ROTY=%3|ROTZ=%4|DZ=%5|CHECKSUM=0|NAME=%6")
330 .arg(id,
331 QString::number(model.rotX, 'f', 6),
332 QString::number(model.rotY, 'f', 6),
333 QString::number(model.rotZ, 'f', 6),
334 QString::number(AltiumCoord::mmToRaw(model.dz)),
335 model.name);
336 QByteArray encoded = metadata.toLatin1();
337 encoded.append('\0');
338 modelWriter.writeInt32(encoded.size());
339 modelWriter.writeBytes(encoded);
340 }
341 ole.writeStream("Library/Models", "Data", modelData);
342
343 // 写入压缩的 STEP 数据（去掉 qCompress 的 4 字节头，Altium 期望原始 zlib 流）
344 for (int i = 0; i < allModels.size(); ++i) {
345 if (!allModels[i].stepData.isEmpty()) {
346 QByteArray compressed = qCompress(allModels[i].stepData, 9);
347 ole.writeStream("Library/Models", QString::number(i), compressed.mid(4));
348 }
349 }

524void AltiumPcbLibWriter::writePad(AltiumBinaryWriter& writer, const AltiumPcbPad& pad, int componentIndex) {
525 writer.writeUInt8(AltiumConstants::PCB_OBJECT_PAD);
526
527 // 子记录 1: Designator
528 writer.writeStringBlock(pad.designator);
529
530 // 子记录 2: PadSubrecord2（空）
531 writer.writeStringBlock("");
532
533 // 子记录 3: Net 字符串
534 writer.writeStringBlock("|&|0");
535
536 // 子记录 4: 标记字节
537 writer.beginBlock();
538 writer.writeUInt8(0);
539 writer.endBlock();
540
541 // 子记录 5: 主焊盘数据
542 writer.beginBlock();
543 {
544 uint8_t layer = pad.isSMD ? pad.layer : AltiumConstants::PCB_LAYER_MULTI;
545 uint16_t flags = encodePrimitiveFlags(pad.isLocked, pad.isTentingTop, pad.isTentingBottom, pad.isKeepout);
546 writeCommonPrimitiveHeader(writer, layer, flags);
547
548 // 位置
549 writer.writeInt32(pad.locationX);
550 writer.writeInt32(pad.locationY);
551
552 // 三层尺寸
553 writer.writeInt32(pad.sizeTopX);
554 writer.writeInt32(pad.sizeTopY);
555 writer.writeInt32(pad.sizeMidX);
556 writer.writeInt32(pad.sizeMidY);
557 writer.writeInt32(pad.sizeBotX);
558 writer.writeInt32(pad.sizeBotY);
559
560 // 孔径
561 writer.writeInt32(pad.holeSize);
562
563 // 三层形状
564 writer.writeUInt8(pad.shapeTop);
565 writer.writeUInt8(pad.shapeMid);
566 writer.writeUInt8(pad.shapeBot);
567
568 // 旋转 + 电镀
569 writer.writeDouble(pad.rotation);
570 writer.writeUInt8(pad.isPlated ? 1 : 0);
571
572 // 补齐主记录固定布局（61 字节之后的字段）
573 writer.writeUInt8(0); // stack mode = Simple
574 writer.writeUInt8(pad.mode);
575 writer.writeUInt8(pad.powerPlaneConnectStyle);
576 writer.writeInt32(pad.reliefAirGapRaw);
577 writer.writeInt32(pad.reliefConductorWidthRaw);
578 writer.writeInt16(pad.reliefEntries);
579 writer.writeInt32(pad.powerPlaneClearanceRaw);
580 writer.writeInt32(pad.powerPlaneReliefExpansionRaw);
581 writer.writeInt32(0); // reserved
582 writer.writeInt32(pad.pasteMaskExpansionRaw);
583 writer.writeInt32(pad.solderMaskExpansionRaw);
584 writer.writeBytes(QByteArray(7, 0)); // reserved
585 writer.writeUInt8(pad.pasteMaskExpansionRaw != 0 ? 2 : 0); // paste mask expansion mode
586 writer.writeUInt8(pad.solderMaskExpansionRaw != 0 ? 2 : 1); // solder mask expansion mode
587 writer.writeUInt8(pad.drillType);
588 writer.writeInt16(0); // reserved
589 writer.writeInt32(0); // reserved
590 writer.writeInt16(0); // jumper ID
591 writer.writeInt16(0); // reserved
592 }
593 writer.endBlock();
594
595 // 子记录 6: 扩展块（各层尺寸/形状覆盖 + 孔元数据）
596 writer.beginBlock();
597 writePadExtendedBlock(writer, pad);
598 writer.endBlock();
599}

608void AltiumPcbLibWriter::writePadExtendedBlock(AltiumBinaryWriter& writer, const AltiumPcbPad& pad) {
609 // 尺寸 X 覆盖：29 层 × 4 字节 = 116 字节
610 for (int i = 0; i < AltiumConstants::PCB_PAD_LAYER_COUNT; ++i) {
611 writer.writeInt32(pad.sizeMidX);
612 }
613 // 尺寸 Y 覆盖：29 层 × 4 字节 = 116 字节
614 for (int i = 0; i < AltiumConstants::PCB_PAD_LAYER_COUNT; ++i) {
615 writer.writeInt32(pad.sizeMidY);
616 }
617 // 形状覆盖：29 个铜层，随后一个保留字节
618 for (int i = 0; i < AltiumConstants::PCB_PAD_LAYER_COUNT; ++i) {
619 writer.writeUInt8(pad.shapeMid);
620 }
621 writer.writeUInt8(0);
622
623 // 孔元数据
624 writer.writeUInt8(pad.holeType);
625 writer.writeInt32(pad.holeSlotLengthRaw);
626 writer.writeDouble(pad.holeRotation);
627
628 // 保留区域
629 writer.writeBytes(QByteArray(32 * 4, 0));
630 writer.writeBytes(QByteArray(32 * 4, 0));
631
632 // 圆角矩形标记
633 bool hasRoundedRect = (pad.shapeTop == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT) ||
634 (pad.shapeMid == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT) ||
635 (pad.shapeBot == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT);
636 writer.writeUInt8(hasRoundedRect ? 1 : 0);
637
638 // 各层形状列表（32 字节：top + 30*mid + bot）
639 writer.writeUInt8(pad.shapeTop);
640 for (int i = 0; i < 30; ++i) {
641 writer.writeUInt8(pad.shapeMid);
642 }
643 writer.writeUInt8(pad.shapeBot);
644
645 // 各层圆角半径百分比（32 字节）
646 for (int i = 0; i < 32; ++i) {
647 writer.writeUInt8(pad.cornerRadiusPercentage);
648 }

698void AltiumPcbLibWriter::writeText(AltiumBinaryWriter& writer, const AltiumPcbText& text, int componentIndex) {
699 writer.writeUInt8(5); // Object ID
700
701 writer.beginBlock();
702 {
703 uint16_t flags = 0x08;
704 writeCommonPrimitiveHeader(writer, text.layer, flags);
705
706 writer.writeInt32(text.locationX);
707 writer.writeInt32(text.locationY);
708 writer.writeInt32(text.height);
709 writer.writeInt16(0); // font ID
710 writer.writeDouble(text.rotation);
711 writer.writeUInt8(text.isMirrored ? 1 : 0);
712 writer.writeInt32(text.strokeWidth);
713 writer.writeUInt8(0); // is comment
714 writer.writeUInt8(0); // is designator
715 writer.writeUInt8(0); // char set
716 writer.writeUInt8(0); // base font type (Stroke)
717
718 // 剩余填充到文本记录总大小（已写字段精确占 44 字节）。
719 constexpr int WRITTEN_TEXT_HEADER_SIZE = 44;
720 QByteArray padding(AltiumConstants::PCB_TEXT_RECORD_SIZE - WRITTEN_TEXT_HEADER_SIZE, 0);
721 // wide string index
722 int wsIdx = addWideString(text.text);
723 int wsBase = AltiumConstants::PCB_TEXT_WS_INDEX_FIELD_OFFSET - WRITTEN_TEXT_HEADER_SIZE;
724 padding[wsBase] = static_cast<char>(wsIdx & 0xFF);
725 padding[wsBase + 1] = static_cast<char>((wsIdx >> 8) & 0xFF);
726 padding[wsBase + 2] = static_cast<char>((wsIdx >> 16) & 0xFF);
727 padding[wsBase + 3] = static_cast<char>((wsIdx >> 24) & 0xFF);
728 // text kind = Stroke
729 padding[AltiumConstants::PCB_TEXT_KIND_FIELD_OFFSET - WRITTEN_TEXT_HEADER_SIZE] = 0;
730 // V7 layer ID
731 uint32_t v7id = toV7LayerId(text.layer);
732 int v7Base = AltiumConstants::PCB_TEXT_V7LAYER_FIELD_OFFSET - WRITTEN_TEXT_HEADER_SIZE;
733 padding[v7Base] = static_cast<char>(v7id & 0xFF);
734 padding[v7Base + 1] = static_cast<char>((v7id >> 8) & 0xFF);
735 padding[v7Base + 2] = static_cast<char>((v7id >> 16) & 0xFF);
736 padding[v7Base + 3] = static_cast<char>((v7id >> 24) & 0xFF);
737
738 writer.writeBytes(padding);
739 }
740 writer.endBlock();
741
742 // 文本字符串块
743 writer.writeStringBlock(text.text);
744}
