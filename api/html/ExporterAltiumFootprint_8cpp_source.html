56AltiumPcbComponent ExporterAltiumFootprint::convertFootprint(const IR::FootprintComponentIR& data,
57 const QString& model3DPath) {
58 AltiumPcbComponent component;
59 component.name = data.name;
60 component.description = data.description;
61 component.height = data.height; // IR height 已经是 mm
62
63 // 转换焊盘
64 for (const IR::FootprintPadIR& pad : data.pads) {
65 component.pads.append(convertPad(pad));
66 }
67
68 // IR 折线必须逐段展开。PcbLib Track 原语只表示一个起点和终点，
69 // 直接取首尾点会把折线错误地压成一条对角线。
70 for (const IR::FootprintTrackIR& track : data.tracks) {
71 for (int i = 1; i < track.points.size(); ++i) {
72 IR::FootprintTrackIR segment = track;
73 segment.points = {track.points[i - 1], track.points[i]};
74 component.tracks.append(convertTrack(segment));
75 }
76 }
77
78 // 独立安装孔用非电镀 MultiLayer pad 表示，使孔径、钻孔属性和图元归属
79 // 都进入同一套已验证的 pad 序列化路径。
80 for (const IR::FootprintHoleIR& hole : data.holes) {
81 AltiumPcbPad pad;
82 pad.locationX = AltiumCoord::mmToRaw(hole.center.x());
83 pad.locationY = AltiumCoord::mmToRaw(hole.center.y());
84 pad.holeSize = AltiumCoord::mmToRaw(hole.radius * 2.0);
85 pad.sizeTopX = pad.sizeMidX = pad.sizeBotX = pad.holeSize;
86 pad.sizeTopY = pad.sizeMidY = pad.sizeBotY = pad.holeSize;
87 pad.shapeTop = pad.shapeMid = pad.shapeBot = AltiumConstants::PCB_PAD_SHAPE_ROUND;
88 pad.layer = AltiumConstants::PCB_LAYER_MULTI;
89 pad.isSMD = false;
90 pad.isPlated = false;
91 pad.isLocked = hole.isLocked;
92 component.pads.append(pad);
93 }
94
95 // 转换圆
96 for (const IR::FootprintCircleIR& circle : data.circles) {
97 component.arcs.append(convertCircle(circle));
98 }
99
100 // 转换弧线
101 for (const IR::FootprintArcIR& arc : data.arcs) {
102 component.arcs.append(convertArc(arc));
103 }
104
105 // 转换矩形
106 for (const IR::FootprintRectangleIR& rect : data.rectangles) {
107 component.fills.append(convertRectangle(rect));
108 }
109
110 // 转换文本
111 for (const IR::FootprintTextIR& text : data.texts) {
112 component.texts.append(convertText(text));
113 }
114
115 // 转换区域
116 for (const IR::FootprintRegionIR& region : data.regions) {
117 component.regions.append(convertSolidRegion(region));
118 }
119
120 // 板框同样是逐段线原语；保留 IR 的层和线宽，不引入导入器特有假设。
121 for (const IR::FootprintOutlineIR& outline : data.outlines) {
122 for (int i = 1; i < outline.points.size(); ++i) {
123 IR::FootprintTrackIR segment;
124 segment.points = {outline.points[i - 1], outline.points[i]};
125 segment.width = outline.strokeWidth;
126 segment.layer = outline.layer;
127 segment.isLocked = outline.isLocked;
128 component.tracks.append(convertTrack(segment));
129 }
130 }
131
132 // 加载 3D 模型
133 QByteArray stepData;
134 QString modelName;
135 if (!model3DPath.isEmpty()) {
136 stepData = loadStepData(model3DPath);
137 modelName = QFileInfo(model3DPath).fileName();
138 } else if (!data.models3d.isEmpty()) {
139 // 从 IR 的 3D 模型中加载 STEP 数据
140 for (const IR::Model3DIR& model3d : data.models3d) {
141 if (model3d.hasStepData()) {
142 AltiumPcbComponent::Model3D model;
143 model.name = model3d.name();
144 model.stepData = model3d.stepData();
145 const auto& rot = model3d.rotation();
146 model.rotX = rot.x;
147 model.rotY = rot.y;
148 model.rotZ = rot.z;
149 const auto& translation = model3d.translation();
150 model.x = AltiumCoord::mmToRaw(translation.x);
151 model.y = AltiumCoord::mmToRaw(translation.y);
152 component.models.append(model);
153 }
154 }
155 if (!component.models.isEmpty()) {
156 stepData = component.models.first().stepData;
157 modelName = component.models.first().name;
158 }
159 }
160
161 // 如果有 3D 模型数据但 models 列表为空，补全
162 if (!stepData.isEmpty() && component.models.isEmpty()) {
163 AltiumPcbComponent::Model3D model;
164 model.name = modelName;
165 model.stepData = stepData;
166 component.models.append(model);
167 }
168
169 // 封装图元和 3D 模型必须经过同一原点平移，避免焊盘与模型错位。
170 centerComponent(component);
171
172 // 为 3D 模型生成 ComponentBody（基于封装包围盒）
173 if (!component.models.isEmpty()) {
174 generateComponentBody(component);
175 }
176
177 return component;
178}

183AltiumPcbPad ExporterAltiumFootprint::convertPad(const IR::FootprintPadIR& pad) {
184 AltiumPcbPad altiumPad;
185 altiumPad.designator = pad.number;
186 altiumPad.locationX = AltiumCoord::mmToRaw(pad.position.x());
187 altiumPad.locationY = AltiumCoord::mmToRaw(pad.position.y());
188
189 // IR size 已经是 mm
190 int width = AltiumCoord::mmToRaw(pad.size.width());
191 int height = AltiumCoord::mmToRaw(pad.size.height());
192
193 // 确定焊盘类型
194 altiumPad.isSMD = pad.isSmd();
195 altiumPad.isPlated = pad.isPlated;
196 altiumPad.isLocked = pad.isLocked;
197
198 if (pad.isThroughHole()) {
199 // 通孔焊盘
200 altiumPad.layer = AltiumConstants::PCB_LAYER_MULTI;
201 altiumPad.holeSize = AltiumCoord::mmToRaw(pad.holeSize);
202
203 // 槽孔检测：holeLength > holeSize 时为槽孔
204 if (pad.holeLength > pad.holeSize + 0.001) {
205 altiumPad.holeType = 2; // Slot
206 altiumPad.holeSlotLengthRaw = AltiumCoord::mmToRaw(pad.holeLength);
207 // 槽孔旋转：长轴方向
208 altiumPad.holeRotation = (pad.size.height() > pad.size.width()) ? 90.0 : 0.0;
209 }
210 } else {
211 // 表贴焊盘：从 IR LayerType 映射到 Altium 层字节
212 altiumPad.layer = static_cast<uint8_t>(AltiumLayerMap::fromLayerTypeToAltium(pad.layer));
213 }
214
215 // 形状：从 IR PadShape 映射到 Altium 焊盘形状字节
216 uint8_t shape = AltiumLayerMap::toAltiumPadShape(pad.shape);
217 altiumPad.shapeTop = shape;
218 altiumPad.shapeMid = shape;
219 altiumPad.shapeBot = shape;
220
221 // 圆角矩形的默认圆角百分比
222 if (shape == AltiumConstants::PCB_PAD_SHAPE_ROUNDED_RECT) {
223 altiumPad.cornerRadiusPercentage = 50;
224 }
225
226 // 尺寸
227 altiumPad.sizeTopX = width;
228 altiumPad.sizeTopY = height;
229 altiumPad.sizeMidX = width;
230 altiumPad.sizeMidY = height;
231 altiumPad.sizeBotX = width;
232 altiumPad.sizeBotY = height;
233
234 altiumPad.rotation = pad.rotation;
235
236 return altiumPad;
237}

242AltiumPcbTrack ExporterAltiumFootprint::convertTrack(const IR::FootprintTrackIR& track) {
243 AltiumPcbTrack altiumTrack;
244 altiumTrack.layer = static_cast<uint8_t>(AltiumLayerMap::fromLayerTypeToAltium(track.layer));
245 altiumTrack.width = AltiumCoord::mmToRaw(track.width);
246
247 // IR points 已经是解析后的 QList<QPointF>，单位 mm
248 if (track.points.size() >= 2) {
249 altiumTrack.startX = AltiumCoord::mmToRaw(track.points.first().x());
250 altiumTrack.startY = AltiumCoord::mmToRaw(track.points.first().y());
251 altiumTrack.endX = AltiumCoord::mmToRaw(track.points.last().x());
252 altiumTrack.endY = AltiumCoord::mmToRaw(track.points.last().y());
253 }
254
255 return altiumTrack;
256}

352void ExporterAltiumFootprint::centerComponent(AltiumPcbComponent& component) {
353 if (component.pads.isEmpty() && component.tracks.isEmpty() && component.arcs.isEmpty() &&
354 component.regions.isEmpty() && component.fills.isEmpty()) {
355 return;
356 }
357
358 // 计算包围盒（使用焊盘位置和走线端点）
359 int minX = INT_MAX, minY = INT_MAX, maxX = INT_MIN, maxY = INT_MIN;
360
361 for (const auto& pad : component.pads) {
362 minX = qMin(minX, pad.locationX);
363 minY = qMin(minY, pad.locationY);
364 maxX = qMax(maxX, pad.locationX);
365 maxY = qMax(maxY, pad.locationY);
366 }
367 for (const auto& track : component.tracks) {
368 minX = qMin(minX, qMin(track.startX, track.endX));
369 minY = qMin(minY, qMin(track.startY, track.endY));
370 maxX = qMax(maxX, qMax(track.startX, track.endX));
371 maxY = qMax(maxY, qMax(track.startY, track.endY));
372 }
373 for (const auto& arc : component.arcs) {
374 minX = qMin(minX, arc.centerX - arc.radius);
375 minY = qMin(minY, arc.centerY - arc.radius);
376 maxX = qMax(maxX, arc.centerX + arc.radius);
377 maxY = qMax(maxY, arc.centerY + arc.radius);
378 }
379 for (const auto& fill : component.fills) {
380 minX = qMin(minX, qMin(fill.corner1X, fill.corner2X));
381 minY = qMin(minY, qMin(fill.corner1Y, fill.corner2Y));
382 maxX = qMax(maxX, qMax(fill.corner1X, fill.corner2X));
383 maxY = qMax(maxY, qMax(fill.corner1Y, fill.corner2Y));
384 }
385 for (const auto& region : component.regions) {
386 for (const QPointF& v : region.vertices) {
387 int vx = static_cast<int>(v.x());
388 int vy = static_cast<int>(v.y());
389 minX = qMin(minX, vx);
390 minY = qMin(minY, vy);
391 maxX = qMax(maxX, vx);
392 maxY = qMax(maxY, vy);
393 }
394 }
395
396 if (minX == INT_MAX)
397 return; // 无图元
398
399 // 计算中心偏移
400 int offsetX = (minX + maxX) / 2;
401 int offsetY = (minY + maxY) / 2;
402
403 // 平移所有图元
404 for (auto& pad : component.pads) {
405 pad.locationX -= offsetX;
406 pad.locationY -= offsetY;
407 }
408 for (auto& track : component.tracks) {
409 track.startX -= offsetX;
410 track.startY -= offsetY;
411 track.endX -= offsetX;
412 track.endY -= offsetY;
413 }
414 for (auto& arc : component.arcs) {
415 arc.centerX -= offsetX;
416 arc.centerY -= offsetY;
417 }
418 for (auto& fill : component.fills) {
419 fill.corner1X -= offsetX;
420 fill.corner1Y -= offsetY;
421 fill.corner2X -= offsetX;
422 fill.corner2Y -= offsetY;
423 }
424 for (auto& region : component.regions) {
425 for (QPointF& v : region.vertices) {
426 v = QPointF(v.x() - offsetX, v.y() - offsetY);
427 }
428 }
429 for (auto& text : component.texts) {
430 text.locationX -= offsetX;
431 text.locationY -= offsetY;
432 }
433 for (auto& model : component.models) {
434 model.x -= offsetX;
435 model.y -= offsetY;
436 }
437}

444void ExporterAltiumFootprint::generateComponentBody(AltiumPcbComponent& component) {
445 if (component.models.isEmpty())
446 return;
447
448 // 计算包围盒（复用 centerComponent 的逻辑）
449 int minX = INT_MAX, minY = INT_MAX, maxX = INT_MIN, maxY = INT_MIN;
450
451 for (const auto& pad : component.pads) {
452 minX = qMin(minX, pad.locationX);
453 minY = qMin(minY, pad.locationY);
454 maxX = qMax(maxX, pad.locationX);
455 maxY = qMax(maxY, pad.locationY);
456 }
457 for (const auto& track : component.tracks) {
458 minX = qMin(minX, qMin(track.startX, track.endX));
459 minY = qMin(minY, qMin(track.startY, track.endY));
460 maxX = qMax(maxX, qMax(track.startX, track.endX));
461 maxY = qMax(maxY, qMax(track.startY, track.endY));
462 }
463 for (const auto& arc : component.arcs) {
464 minX = qMin(minX, arc.centerX - arc.radius);
465 minY = qMin(minY, arc.centerY - arc.radius);
466 maxX = qMax(maxX, arc.centerX + arc.radius);
467 maxY = qMax(maxY, arc.centerY + arc.radius);
468 }
469 for (const auto& fill : component.fills) {
470 minX = qMin(minX, qMin(fill.corner1X, fill.corner2X));
471 minY = qMin(minY, qMin(fill.corner1Y, fill.corner2Y));
472 maxX = qMax(maxX, qMax(fill.corner1X, fill.corner2X));
473 maxY = qMax(maxY, qMax(fill.corner1Y, fill.corner2Y));
474 }
475 for (const auto& region : component.regions) {
476 for (const QPointF& v : region.vertices) {
477 minX = qMin(minX, static_cast<int>(v.x()));
478 minY = qMin(minY, static_cast<int>(v.y()));
479 maxX = qMax(maxX, static_cast<int>(v.x()));
480 maxY = qMax(maxY, static_cast<int>(v.y()));
481 }
482 }
483
484 if (minX == INT_MAX)
485 return; // 无图元，无法生成轮廓
486
487 // 生成稳定的模型 ID（基于封装名称）
488 QString seed = component.name + "|body";
489 QByteArray seedBytes = seed.toUtf8();
490 uint64_t hash = 0xCBF29CE484222325ULL;
491 for (char byte : seedBytes) {
492 hash ^= static_cast<uint8_t>(byte);
493 hash *= 0x100000001B3ULL;
494 }
495 QString modelId = QString("{%1-%2-%3-%4-%5}")
496 .arg((hash >> 32) & 0xFFFFFFFF, 8, 16, QChar('0'))
497 .arg((hash >> 16) & 0xFFFF, 4, 16, QChar('0'))
498 .arg(hash & 0xFFFF, 4, 16, QChar('0'))
499 .arg(((~hash) >> 48) & 0xFFFF, 4, 16, QChar('0'))
500 .arg((~hash) & 0xFFFFFFFFFFFFLL, 12, 16, QChar('0'))
501 .toUpper();
502 component.models.first().id = modelId;
503
504 AltiumPcbComponentBody body;
505 body.modelId = modelId;
506 body.modelName = component.models.first().name;
507 body.model2dRotX = component.models.first().x;
508 body.model2dRotY = component.models.first().y;
509 body.model3dRotX = component.models.first().rotX;
510 body.model3dRotY = component.models.first().rotY;
511 body.model3dRotZ = component.models.first().rotZ;
512 body.model3dDzRaw = AltiumCoord::mmToRaw(component.models.first().dz);
513 body.overallHeightRaw = AltiumCoord::mmToRaw(qMax(component.height, 0.2));
514
515 // 矩形轮廓
516 body.outline = {QPointF(minX, minY), QPointF(maxX, minY), QPointF(maxX, maxY), QPointF(minX, maxY)};
517
518 component.bodies.append(body);
519}
