53QString SymbolGraphicsGenerator::generatePin(const IR::SymbolPinIR& pin) const {
54 QString content;
55
56 // 坐标已为 mm，直接计算相对位置
57 double x = pin.position.x() - m_originX;
58 double y = pin.position.y() - m_originY;
59
60 // 引脚长度（已为 mm）
61 double length = pin.length;
62
63 // 确保引脚长度为正数（KiCad 引脚长度必须是正数）
64 length = std::abs(length);
65
66 // 确保引脚长度不为 0
67 if (length < 0.01) {
68 length = 2.54; // 默认引脚长度（100mil）
69 }
70
71 // 使用 IR 映射获取 KiCad 引脚电气类型
72 const char* kicadPinType = KiCadTypeMap::toKicadPinType(pin.electricalType);
73
74 // 引脚样式：从 IR 语义层获取
75 QString kicadPinStyle = "line";
76 if (pin.style.inverted && pin.style.clock) {
77 kicadPinStyle = "inverted_clock";
78 } else if (pin.style.inverted) {
79 kicadPinStyle = "inverted";
80 } else if (pin.style.clock) {
81 kicadPinStyle = "clock";
82 }
83
84 // 处理引脚名称和编号
85 QString pinName = pin.name;
86 pinName.replace(" ", "");
87 QString pinNumber = pin.designator;
88 pinNumber.replace(" ", "");
89
90 // 如果引脚名称为空，使用引脚编号
91 if (pinName.isEmpty()) {
92 pinName = pinNumber;
93 }
94
95 const bool useOriginalNamePosition = pin.hasNamePosition && !pinName.isEmpty();
96 const bool useOriginalNumberPosition = pin.hasNumberPosition && !pinNumber.isEmpty();
97 const QString nameHide = useOriginalNamePosition ? QStringLiteral(" hide") : QString();
98 const QString numberHide = useOriginalNumberPosition ? QStringLiteral(" hide") : QString();
99
100 // 使用 IR 方向枚举获取 KiCad 角度
101 double kicadOrientation = KiCadTypeMap::toKicadAngle(pin.direction);
102
103 content += QString(" (pin %1 %2\n").arg(kicadPinType, kicadPinStyle);
104 content += QString(" (at %1 %2 %3)\n").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2).arg(kicadOrientation, 0, 'f', 0);
105 content += QString(" (length %1)\n").arg(length, 0, 'f', 2);
106 if (useOriginalNamePosition) {
107 content +=
108 QString(" (name \"%1\" (effects (font (size 1.27 1.27) (thickness 0))%2))\n").arg(pinName, nameHide);
109 } else {
110 content += QString(" (name \"%1\" (effects (font (size 1.27 1.27) (thickness 0) )))\n").arg(pinName);
111 }
112 if (useOriginalNumberPosition) {
113 content += QString(" (number \"%1\" (effects (font (size 1.27 1.27) (thickness 0))%2))\n")
114 .arg(pinNumber, numberHide);
115 } else {
116 content += QString(" (number \"%1\" (effects (font (size 1.27 1.27) (thickness 0) )))\n").arg(pinNumber);
117 }
118 content += " )\n";
119
120 auto kicadJustification = [](const QString& anchor) {
121 const QString normalized = anchor.trimmed().toLower();
122 if (normalized == QStringLiteral("start") || normalized == QStringLiteral("left"))
123 return QStringLiteral(" (justify left)");
124 if (normalized == QStringLiteral("end") || normalized == QStringLiteral("right"))
125 return QStringLiteral(" (justify right)");
126 if (normalized == QStringLiteral("top"))
127 return QStringLiteral(" (justify top)");
128 if (normalized == QStringLiteral("bottom"))
129 return QStringLiteral(" (justify bottom)");
130 return QString();
131 };
132 auto appendOriginalText =
133 [&](const QString& text, const QPointF& position, double rotation, const QString& anchor, double fontSizeMm) {
134 double textRotation = rotation;
135 if (textRotation != 0.0)
136 textRotation = 360.0 - textRotation;
137 const double size = fontSizeMm > 0.0 ? fontSizeMm : 1.27;
138 content += " (text\n";
139 content += QString(" \"%1\"\n").arg(text);
140 content += QString(" (at %1 %2 %3)\n")
141 .arg(position.x() - m_originX, 0, 'f', 2)
142 .arg(position.y() - m_originY, 0, 'f', 2)
143 .arg(textRotation, 0, 'f', 0);
144 content += QString(" (effects (font (size %1 %1) (thickness 0.1))%2)\n")
145 .arg(size, 0, 'f', 2)
146 .arg(kicadJustification(anchor));
147 content += " )\n";
148 };
149 if (useOriginalNamePosition)
150 appendOriginalText(pinName, pin.namePosition, pin.nameRotation, pin.nameAnchor, pin.nameFontSizeMm);
151 if (useOriginalNumberPosition)
152 appendOriginalText(pinNumber, pin.numberPosition, pin.numberRotation, pin.numberAnchor, pin.numberFontSizeMm);
153
154 return content;
155}

225QString SymbolGraphicsGenerator::generateEllipse(const IR::SymbolEllipseIR& ellipse) const {
226 QString content;
227
228 // 坐标已为 mm，计算相对位置
229 double cx = ellipse.center.x() - m_originX;
230 double cy = ellipse.center.y() - m_originY;
231 double radiusX = ellipse.radiusX;
232 double radiusY = ellipse.radiusY;
233 double strokeWidth = ellipse.strokeWidth;
234
235 // 如果是圆形（radiusX 等于 radiusY），使用 circle 元素
236 if (qAbs(radiusX - radiusY) < 0.01) {
237 content += " (circle\n";
238 content += QString(" (center %1 %2)\n").arg(cx, 0, 'f', 2).arg(cy, 0, 'f', 2);
239 content += QString(" (radius %1)\n").arg(radiusX, 0, 'f', 2);
240 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);
241 content += " (fill (type none))\n";
242 content += " )\n";
243 } else {
244 // 椭圆：使用 32 段折线近似
245 content += " (polyline\n";
246 content += " (pts";
247
248 const int segments = 32;
249 for (int i = 0; i <= segments; ++i) {
250 double angle = 2.0 * M_PI * i / segments;
251 double x = cx + radiusX * std::cos(angle);
252 double y = cy + radiusY * std::sin(angle);
253 content += QString(" (xy %1 %2)").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2);
254 }
255
256 content += ")\n";
257 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);
258
259 if (ellipse.isFilled) {
260 content += " (fill (type background))\n";
261 } else {
262 content += " (fill (type none))\n";
263 }
264 content += " )\n";
265 }
266
267 return content;
268}

270QString SymbolGraphicsGenerator::generatePolygon(const IR::SymbolPolygonIR& polygon) const {
271 QString content;
272 double strokeWidth = polygon.strokeWidth;
273
274 // IR 中坐标已解析为 QList<QPointF>
275 if (polygon.points.size() >= 2) {
276 // KiCad V6 不支持 polygon 元素，使用 polyline 代替
277 content += " (polyline\n";
278 content += " (pts";
279 QString firstPoint;
280 QString lastPoint;
281 for (const QPointF& pt : polygon.points) {
282 double x = pt.x() - m_originX;
283 double y = pt.y() - m_originY;
284 QString point = QString(" (xy %1 %2)").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2);
285
286 // 避免重复点
287 if (point != lastPoint) {
288 content += point;
289 if (firstPoint.isEmpty()) {
290 firstPoint = point;
291 }
292 lastPoint = point;
293 }
294 }
295 // 多边形总是重复第一个点以闭合
296 if (!firstPoint.isEmpty() && firstPoint != lastPoint) {
297 content += firstPoint;
298 }
299 content += ")\n";
300 content += QString(" (stroke (width %1) (type default))\n").arg(strokeWidth, 0, 'f', 3);
301 if (polygon.isFilled) {
302 content += " (fill (type background))\n";
303 } else {
304 content += " (fill (type none))\n";
305 }
306 content += " )\n";
307 }
308
309 return content;

397QString SymbolGraphicsGenerator::generateText(const IR::SymbolTextIR& text) const {
398 QString content;
399
400 // 坐标已为 mm，计算相对位置
401 double x = text.position.x() - m_originX;
402 double y = text.position.y() - m_originY;
403
404 // IR 中 fontSizeMm 已从 pt 转换；部分 EasyEDA 文本记录未提供字号，
405 // 使用与平台属性一致的默认字号，避免生成 0.00 的不可见文本。
406 double fontSize = text.fontSizeMm > 0.0 ? text.fontSizeMm : 1.27;
407
408 // 处理粗体和斜体（IR 中 italic 已为 bool）
409 QString fontStyle = "";
410 if (text.bold) {
411 fontStyle += "bold ";
412 }
413 if (text.italic) {
414 fontStyle += "italic ";
415 }
416 if (fontStyle.isEmpty()) {
417 fontStyle = "normal";
418 } else {
419 fontStyle = fontStyle.trimmed();
420 }
421
422 // 处理旋转角度
423 double rotation = text.rotation;
424 if (rotation != 0) {
425 rotation = 360 - rotation;
426 }
427
428 // 处理可见性
429 QString hide = text.visible ? "" : "hide";
430
431 // 转义文本内容
432 QString textContent = text.text;
433 textContent.replace(" ", "");
434 if (textContent.isEmpty()) {
435 textContent = "~";
436 }
437
438 // 生成文本元素
439 content += " (text\n";
440 content += QString(" \"%1\"\n").arg(textContent);
441 content += QString(" (at %1 %2 %3)\n").arg(x, 0, 'f', 2).arg(y, 0, 'f', 2).arg(rotation, 0, 'f', 0);
442 content += QString(" (effects (font (size %1 %2) (thickness 0.1) %3) %4)\n")
443 .arg(fontSize, 0, 'f', 2)
444 .arg(fontSize, 0, 'f', 2)
445 .arg(fontStyle)
446 .arg(hide);
447 content += " )\n";
448
449 return content;
450}
