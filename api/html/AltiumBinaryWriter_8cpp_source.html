78void AltiumBinaryWriter::endBlock() {
79 if (m_blockPositions.isEmpty()) {
80 return;
81 }
82
83 int blockStart = m_blockPositions.takeLast();
84 int payloadSize = m_buffer.size() - blockStart - 4;
85 uint8_t flags = static_cast<uint8_t>(m_buffer[blockStart]);
86
87 // 回填大小：(flags << 24) | payload_size
88 uint32_t sizeHeader = (static_cast<uint32_t>(flags) << 24) | (static_cast<uint32_t>(payloadSize) & 0x00FFFFFF);
89 m_buffer[blockStart] = static_cast<char>(sizeHeader & 0xFF);
90 m_buffer[blockStart + 1] = static_cast<char>((sizeHeader >> 8) & 0xFF);
91 m_buffer[blockStart + 2] = static_cast<char>((sizeHeader >> 16) & 0xFF);
92 m_buffer[blockStart + 3] = static_cast<char>((sizeHeader >> 24) & 0xFF);
93}

128 QString paramStr;

130 for (const QString& preferred : {QStringLiteral("HEADER"), QStringLiteral("RECORD"), QStringLiteral("PATTERN")}) {
131 if (params.contains(preferred))

134 for (auto it = params.constBegin(); it != params.constEnd(); ++it) {

154void AltiumBinaryWriter::writeCStringParameterBlockUtf8(const QMap<QString, QString>& params) {
155 QByteArray encoded;
156 // Altium 的参数块虽然语义上是键值集合，但记录识别器会读取首个
157 // 参数来判断记录类型。QMap 的字典序会把 ALLPINCOUNT/AREACOLOR
158 // 放在 RECORD/HEADER 前面，从而使合法数据被当作未知记录。保持少量
159 // 协议首字段的稳定顺序，其余字段仍使用 QMap 的确定性顺序。
160 QList<QString> orderedKeys;
161 for (const QString& preferred : {QStringLiteral("HEADER"), QStringLiteral("RECORD"), QStringLiteral("PATTERN")}) {
162 if (params.contains(preferred))
163 orderedKeys.append(preferred);
164 }
165 for (auto it = params.constBegin(); it != params.constEnd(); ++it) {
166 if (!orderedKeys.contains(it.key()))
167 orderedKeys.append(it.key());
168 }
169
170 for (const QString& keyName : orderedKeys) {
171 auto it = params.constFind(keyName);
172 if (it == params.constEnd())
173 continue;
174 const QByteArray key = it.key().toLatin1();
175 QString value = it.value();
176 value.replace('|', ' ');
177 value.replace(QChar::Null, ' ');
178 const QByteArray ansiValue = value.toLatin1();
179 encoded += '|';
180 encoded += key;
181 encoded += '=';
182 encoded += ansiValue;
183
184 // %UTF8% 参数的值必须是原始 UTF-8 字节，而不是再次 Latin-1 转码。
185 if (QString::fromLatin1(ansiValue) != value) {
186 encoded += "|%UTF8%";
187 encoded += key;
188 encoded += '=';
189 encoded += value.toUtf8();
190 }
191 }
192 encoded += '|';
193 writeCStringParameterBlockRaw(encoded);
194}
