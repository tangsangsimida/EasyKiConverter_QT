24static void computeBBox(const IR::FootprintComponentIR& footprint,
25 double& outMinX,
26 double& outMinY,
27 double& outMaxX,
28 double& outMaxY) {
29 outMinX = std::numeric_limits<double>::max();
30 outMinY = std::numeric_limits<double>::max();
31 outMaxX = std::numeric_limits<double>::lowest();
32 outMaxY = std::numeric_limits<double>::lowest();
33
34 for (const auto& pad : footprint.pads) {
35 outMinX = qMin(outMinX, pad.position.x() - pad.size.width() / 2.0);
36 outMinY = qMin(outMinY, pad.position.y() - pad.size.height() / 2.0);
37 outMaxX = qMax(outMaxX, pad.position.x() + pad.size.width() / 2.0);
38 outMaxY = qMax(outMaxY, pad.position.y() + pad.size.height() / 2.0);
39 }
40 for (const auto& track : footprint.tracks) {
41 for (const QPointF& pt : track.points) {
42 outMinX = qMin(outMinX, pt.x());
43 outMinY = qMin(outMinY, pt.y());
44 outMaxX = qMax(outMaxX, pt.x());
45 outMaxY = qMax(outMaxY, pt.y());
46 }
47 }
48 for (const auto& hole : footprint.holes) {
49 outMinX = qMin(outMinX, hole.center.x() - hole.radius);
50 outMinY = qMin(outMinY, hole.center.y() - hole.radius);
51 outMaxX = qMax(outMaxX, hole.center.x() + hole.radius);
52 outMaxY = qMax(outMaxY, hole.center.y() + hole.radius);
53 }
54 for (const auto& circle : footprint.circles) {
55 outMinX = qMin(outMinX, circle.center.x() - circle.radius);
56 outMinY = qMin(outMinY, circle.center.y() - circle.radius);
57 outMaxX = qMax(outMaxX, circle.center.x() + circle.radius);
58 outMaxY = qMax(outMaxY, circle.center.y() + circle.radius);
59 }
60 for (const auto& rect : footprint.rectangles) {
61 outMinX = qMin(outMinX, rect.bounds.left());
62 outMinY = qMin(outMinY, rect.bounds.top());
63 outMaxX = qMax(outMaxX, rect.bounds.right());
64 outMaxY = qMax(outMaxY, rect.bounds.bottom());
65 }
66 for (const auto& arc : footprint.arcs) {
67 outMinX = qMin(outMinX, arc.center.x() - arc.radius);
68 outMinY = qMin(outMinY, arc.center.y() - arc.radius);
69 outMaxX = qMax(outMaxX, arc.center.x() + arc.radius);
70 outMaxY = qMax(outMaxY, arc.center.y() + arc.radius);
71 }
72 for (const auto& text : footprint.texts) {
73 outMinX = qMin(outMinX, text.position.x());
74 outMinY = qMin(outMinY, text.position.y());
75 outMaxX = qMax(outMaxX, text.position.x());
76 outMaxY = qMax(outMaxY, text.position.y());
77 }
78 for (const auto& region : footprint.regions) {
79 for (const QPointF& pt : region.vertices) {
80 outMinX = qMin(outMinX, pt.x());
81 outMinY = qMin(outMinY, pt.y());
82 outMaxX = qMax(outMaxX, pt.x());
83 outMaxY = qMax(outMaxY, pt.y());
84 }
85 }
86}

137bool ExporterFootprint::exportFootprintLibrary(const QList<IR::FootprintComponentIR>& footprints,
138 const QString& libName,
139 const QString& filePath,
140 bool preferWrl,
141 bool exportStep,
142 const QString& libraryDescription,
143 const QString& libraryKeywords,
144 bool useAbsolutePaths,
145 const QString& model3DBaseDir) {
146 qDebug() << "=== Export Footprint Library ===";
147 qDebug() << "Library name:" << libName;
148 qDebug() << "Output path:" << filePath;
149 qDebug() << "Footprint count:" << footprints.count();
150 qDebug() << "Library description:" << libraryDescription;
151 qDebug() << "Library keywords:" << libraryKeywords;
152
153 QDir libDir(filePath);
154
155 if (!libDir.exists()) {
156 qDebug() << "Creating footprint library directory:" << filePath;
157 if (!libDir.mkpath(".")) {
158 qWarning() << "Failed to create footprint library directory:" << filePath;
159 return false;
160 }
161 }
162
163 QSet<QString> existingFootprintNames;
164 QStringList existingFiles = libDir.entryList(QStringList("*.kicad_mod"), QDir::Files);
165 for (const QString& fileName : existingFiles) {
166 QString fpName = fileName;
167 fpName.remove(".kicad_mod");
168 existingFootprintNames.insert(fpName);
169 qDebug() << "Found existing footprint:" << fpName;
170 }
171
172 qDebug() << "Existing footprints count:" << existingFootprintNames.count();
173
174 const QString safeLibName = PathSecurity::sanitizeFilename(libName);
175 const QString resolvedBaseDir = model3DBaseDir.isEmpty() ? QFileInfo(filePath).absolutePath() : model3DBaseDir;
176
177 int exportedCount = 0;
178 int overwrittenCount = 0;
179 for (const IR::FootprintComponentIR& footprint : footprints) {
180 QString fpName = footprint.name;
181 QString fileName = fpName + ".kicad_mod";
182 QString fullPath = libDir.filePath(fileName);
183
184 bool exists = existingFootprintNames.contains(fpName);
185 if (exists) {
186 qDebug() << "Overwriting existing footprint:" << fpName;
187 overwrittenCount++;
188 } else {
189 qDebug() << "Exporting new footprint:" << fpName;
190 exportedCount++;
191 }
192
193 // 确定 3D 模型路径
194 // 只要 models3d 非空且有名称即视为有 3D 模型引用（不依赖 isValid()，
195 // 因为 isValid() 要求有实际 OBJ/STEP 数据，而路径生成仅需名称）
196 bool hasModel3D = !footprint.models3d.isEmpty() && !footprint.models3d.first().name().isEmpty();
197 QString modelName;
198 if (hasModel3D) {
199 modelName = footprint.models3d.first().name();
200 if (modelName.isEmpty()) {
201 modelName = fpName;
202 }
203 modelName = PathSecurity::sanitizeFilename(modelName);
204 }
205
206 QString content;
207 const bool useWrl = preferWrl && hasModel3D;
208 const bool useStep = exportStep && hasModel3D;
209
210 if (useWrl && useStep) {
211 QString wrlPath =
212 buildModel3DPath(safeLibName, modelName, QStringLiteral("wrl"), useAbsolutePaths, resolvedBaseDir);
213 QString stepPath =
214 buildModel3DPath(safeLibName, modelName, QStringLiteral("step"), useAbsolutePaths, resolvedBaseDir);
215 content = generateFootprintContent(footprint, wrlPath, stepPath, libraryDescription, libraryKeywords);
216 } else if (useWrl) {
217 QString wrlPath =
218 buildModel3DPath(safeLibName, modelName, QStringLiteral("wrl"), useAbsolutePaths, resolvedBaseDir);
219 content = generateFootprintContent(footprint, wrlPath, libraryDescription, libraryKeywords);
220 } else if (useStep) {
221 QString stepPath =
222 buildModel3DPath(safeLibName, modelName, QStringLiteral("step"), useAbsolutePaths, resolvedBaseDir);
223 content = generateFootprintContent(footprint, stepPath, libraryDescription, libraryKeywords);
224 } else {
225 content = generateFootprintContent(footprint, QString(), libraryDescription, libraryKeywords);
226 }
227
228 QFile file(fullPath);
229 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
230 qWarning() << "Failed to open file for writing:" << fullPath;
231 continue;
232 }
233
234 QTextStream out(&file);
235 out.setEncoding(QStringConverter::Utf8);
236 out << content;
237 file.close();
238
239 qDebug() << " Exported to:" << fullPath;
240 }
241
242 qDebug() << "Export complete - New:" << exportedCount << "Overwritten:" << overwrittenCount;
243 qDebug() << "Footprint library exported to:" << filePath;
244
245 return true;
246}

272void ExporterFootprint::generateFootprintBaseContent(const IR::FootprintComponentIR& footprint,
273 QString& content,
274 double& outOriginX,
275 double& outOriginY,
276 const QString& libraryDescription,
277 const QString& libraryKeywords) const {
278 content += QString("(footprint easykiconverter:%1\n").arg(footprint.name);
279 content += " (version 20221018)\n";
280
281 // 导出封装描述 (descr)
282 const QString fpDescription = footprint.description.trimmed();
283 if (!fpDescription.isEmpty()) {
284 content += QString(" (descr \"%1\")\n").arg(fpDescription);
285 }
286
287 // 导出封装标签 (tags)
288 if (!libraryKeywords.isEmpty()) {
289 content += QString(" (tags \"%1\")\n").arg(libraryKeywords);
290 }
291
292 // 判断是否有通孔焊盘
293 if (footprint.hasThroughHolePads()) {
294 content += " (attr through_hole)\n";
295 } else {
296 content += " (attr smd)\n";
297 }
298
299 // 计算边界框（mm 单位）
300 double bboxMinX, bboxMinY, bboxMaxX, bboxMaxY;
301 computeBBox(footprint, bboxMinX, bboxMinY, bboxMaxX, bboxMaxY);
302
303 // 如果没有有效的边界框，使用默认值
304 bool hasValidBBox = (bboxMinX <= bboxMaxX && bboxMinY <= bboxMaxY);
305 if (!hasValidBBox) {
306 bboxMinX = bboxMinY = 0.0;
307 bboxMaxX = bboxMaxY = 1.0;
308 }
309
310 // 计算边界框中心作为原点
311 double originX = (bboxMinX + bboxMaxX) / 2.0;
312 double originY = (bboxMinY + bboxMaxY) / 2.0;
313
314 // 计算 pad Y 范围（相对于原点）
315 double yLow = 0;
316 double yHigh = 0;
317 if (!footprint.pads.isEmpty()) {
318 yLow = footprint.pads.first().position.y() - originY;
319 yHigh = yLow;
320 for (const IR::FootprintPadIR& pad : footprint.pads) {
321 double padY = pad.position.y() - originY;
322 if (padY < yLow)
323 yLow = padY;
324 if (padY > yHigh)
325 yHigh = padY;
326 }
327 }
328
329 content += QString(" (fp_text reference REF** (at 0 %1) (layer F.SilkS)\n").arg(yLow - 4 * IR::EASYEDA_PX_TO_MM);
330 content += " (effects (font (size 1 1) (thickness 0.15)))\n";
331 content += " )\n";
332
333 content += QString(" (fp_text value %1 (at 0 %2) (layer F.Fab)\n")
334 .arg(footprint.name)
335 .arg(yHigh + 4 * IR::EASYEDA_PX_TO_MM);
336 content += " (effects (font (size 1 1) (thickness 0.15)))\n";
337 content += " )\n";
338
339 for (const IR::FootprintTrackIR& track : footprint.tracks) {
340 content += m_graphicsGenerator.generateTrack(track, originX, originY);
341 }
342 for (const IR::FootprintRectangleIR& rect : footprint.rectangles) {
343 content += m_graphicsGenerator.generateRectangle(rect, originX, originY);
344 }
345
346 for (const IR::FootprintPadIR& pad : footprint.pads) {
347 content += m_graphicsGenerator.generatePad(pad, originX, originY);
348 }
349
350 for (const IR::FootprintHoleIR& hole : footprint.holes) {
351 content += m_graphicsGenerator.generateHole(hole, originX, originY);
352 }
353
354 for (const IR::FootprintCircleIR& circle : footprint.circles) {
355 content += m_graphicsGenerator.generateCircle(circle, originX, originY);
356 }
357
358 for (const IR::FootprintArcIR& arc : footprint.arcs) {
359 content += m_graphicsGenerator.generateArc(arc, originX, originY);
360 }
361
362 for (const IR::FootprintTextIR& text : footprint.texts) {
363 content += m_graphicsGenerator.generateText(text, originX, originY);
364 }
365
366 bool hasCourtYard = false;
367 for (const IR::FootprintRegionIR& region : footprint.regions) {
368 QString regionContent = m_graphicsGenerator.generateSolidRegion(region, originX, originY);
369 content += regionContent;
370 if (region.layer == IR::LayerType::KeepOut) {
371 hasCourtYard = true;
372 }
373 }
374
375 if (!hasCourtYard && footprint.shouldGenerateCourtyard && hasValidBBox) {
376 double x1 = std::floor((bboxMinX - originX) * 100.0) / 100.0;
377 double y1 = std::floor((bboxMinY - originY) * 100.0) / 100.0;
378 double x2 = std::floor((bboxMaxX - originX) * 100.0) / 100.0;
379 double y2 = std::floor((bboxMaxY - originY) * 100.0) / 100.0;
380 content += m_graphicsGenerator.generateCourtyardFromBBox(x1, y1, x2, y2);
381 qWarning() << "Warning: No courtyard found, generated from BBox";
382 }
383
384 outOriginX = originX;
385 outOriginY = originY;
386}
