68 auto convertPins = [&](const QList<SymbolPin>& pins, double originX, double originY, int partIdx = 0) {
69 for (const auto& pin : pins) {
70 SymbolPinIR pir;
71 pir.name = pin.name.text;
72 pir.designator = pin.settings.spicePinNumber;
73 pir.position = GeometryNormalizer::transformPoint(pin.settings.posX, pin.settings.posY, originX, originY);
74 const bool hasNamePosition = pin.name.isDisplayed && !pin.name.text.isEmpty() &&
75 std::isfinite(pin.name.posX) && std::isfinite(pin.name.posY);
76 pir.namePosition = hasNamePosition

81 pir.nameFontSizeMm = pin.name.fontSize * EASYEDA_PX_TO_MM;
82 pir.hasNamePosition = hasNamePosition;
83 const bool hasNumberPosition = pin.number.isDisplayed && !pin.number.text.isEmpty() &&
84 std::isfinite(pin.number.posX) && std::isfinite(pin.number.posY);
85 pir.numberPosition = hasNumberPosition ? GeometryNormalizer::transformPoint(
86 pin.number.posX, pin.number.posY, originX, originY)
87 : pir.position;
88 pir.numberRotation = pin.number.rotation;
89 pir.numberAnchor = pin.number.textAnchor;
90 pir.numberFontSizeMm = pin.number.fontSize * EASYEDA_PX_TO_MM;
91 pir.hasNumberPosition = hasNumberPosition;
92
93 // 从 pinPath SVG 解析引脚长度
94 const QString pathStr = pin.pinPath.path;
95 static const QRegularExpression lengthExpression(QStringLiteral(R"([hHvV]\s*([-+]?(?:\d+\.?\d*|\.\d+)))"));
96 const QRegularExpressionMatch lengthMatch = lengthExpression.match(pathStr);
97 if (lengthMatch.hasMatch()) {
98 bool ok = false;
99 const double len = lengthMatch.captured(1).toDouble(&ok);
100 if (ok)
101 pir.length = qAbs(len) * EASYEDA_PX_TO_MM;
102 }
103
104 pir.direction = EasyedaPinTypeMap::toPinDirection(pin.settings.rotation);
105 pir.electricalType = EasyedaPinTypeMap::toPinElectricalType(static_cast<int>(pin.settings.type));
106
107 // 显示控制
108 pir.showName = pin.name.isDisplayed;
109 pir.showDesignator = pin.settings.isDisplayed;
110 pir.display.showName = pin.name.isDisplayed;
111 pir.display.showDesignator = pin.settings.isDisplayed;
112
113 // 引脚样式（语义层）
114 pir.hasDot = pin.dot.isDisplayed;
115 pir.hasClock = pin.clock.isDisplayed;
116 pir.style.inverted = pin.dot.isDisplayed;
117 pir.style.clock = pin.clock.isDisplayed;
118 if (pin.dot.isDisplayed && pin.clock.isDisplayed) {
119 pir.style.decoration = PinDecoration::InvertedClock;
120 } else if (pin.dot.isDisplayed) {
121 pir.style.decoration = PinDecoration::Dot;
122 } else if (pin.clock.isDisplayed) {
123 pir.style.decoration = PinDecoration::Clock;
124 }

126 pir.partIndex = partIdx;
127 ir.pins.append(pir);
